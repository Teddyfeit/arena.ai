trace.glsl — the actual heart of this page 
vec3 trace(vec3 ro, vec3 rd) {
  vec3 throughput = vec3(1.0), radiance = vec3(0.0);
  for (int bounce = 0; bounce < 3; bounce++) {
    Hit h = intersectScene(ro, rd);      // spheres · box · floor
    if (h.t >= MAX_DIST) { radiance += throughput * sky(rd); break; }
    radiance   += throughput * shade(h, ro, rd);  // key + 2 shadowed accent lights
    throughput *= fresnel(h, rd) * mix(vec3(1.0), h.albedo, 0.72);
    ro = h.pos + h.normal * 0.0025;
    rd = reflect(rd, h.normal);           // ← the RTX part
  }
  return aces(radiance * 1.15);
}
